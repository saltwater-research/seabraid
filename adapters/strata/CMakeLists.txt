cmake_minimum_required(VERSION 3.16)
project(seabraid_strata_adapter LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The Strata v7 adapter requires Linux shared-memory APIs")
endif()

if(DEFINED SEABRAID_EMBED_STRATA_SOURCE)
  set(STRATA_SOURCE "${SEABRAID_EMBED_STRATA_SOURCE}")
  set(STRATA_GEOMETRY_INCLUDE "${SEABRAID_EMBED_STRATA_GEOMETRY}")
  set(STRATA_WINDOW_MAX_T "${SEABRAID_EMBED_STRATA_WINDOW}")
else()
  set(STRATA_SOURCE "" CACHE PATH "Strata source checkout (pinned upstream; the placement targets need the engine patches)")
  set(STRATA_GEOMETRY_INCLUDE "" CACHE PATH "Optional Strata generated include directory")
  set(STRATA_WINDOW_MAX_T "" CACHE STRING "Helper's window limit; empty uses the Strata header default")
endif()
# WHY geometry.hpp and not the channel: the helper wire is this adapter's own (wire/, see
# wire/strata/seabraid/channel_layout.hpp); from Strata the adapter needs only engine headers it does not own.
if(NOT EXISTS "${STRATA_SOURCE}/include/strata/core/geometry.hpp")
  message(FATAL_ERROR "Set STRATA_SOURCE to a Strata checkout (include/strata/core/geometry.hpp not found)")
endif()
set(SEABRAID_STRATA_WIRE "${CMAKE_CURRENT_SOURCE_DIR}/wire")
if(STRATA_GEOMETRY_INCLUDE AND NOT EXISTS "${STRATA_GEOMETRY_INCLUDE}/strata/core/geometry_selected.hpp")
  message(FATAL_ERROR "STRATA_GEOMETRY_INCLUDE must contain strata/core/geometry_selected.hpp")
endif()

# Hardened compile and link flags (seabraid_harden): one owner, shared with compat's seabraid-helper.
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/seabraid_harden.cmake")
include(CheckCXXSymbolExists)
set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
check_cxx_symbol_exists(pidfd_spawn spawn.h SEABRAID_HAVE_PIDFD_SPAWN)
unset(CMAKE_REQUIRED_DEFINITIONS)

# The diagnostics seam is its own library so the placement layer and the standalone
# v7 fixtures share one sink without linking each other.
add_library(seabraid_strata_diag STATIC diag.cpp)
set_target_properties(seabraid_strata_diag PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(seabraid_strata_diag PUBLIC cxx_std_17)
target_include_directories(seabraid_strata_diag PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "../../include")
seabraid_harden(seabraid_strata_diag LIB)

# The opt-in move ledger and latency trace (SEABRAID_TRACE): diag and the placement ABI header only, so it builds
# without Strata or the Rust runtime and the placement layer, the runtime and its test share one copy.
add_library(seabraid_strata_trace STATIC placement_trace.cpp)
set_target_properties(seabraid_strata_trace PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(seabraid_strata_trace PUBLIC cxx_std_17)
target_include_directories(seabraid_strata_trace PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "../../include")
target_link_libraries(seabraid_strata_trace PUBLIC seabraid_strata_diag)
seabraid_harden(seabraid_strata_trace LIB)

# File-reference expert moves (FR-E): the manifest and the per-helper decision.  Diag, the wire and Strata's RungIO header only, so it
# builds and tests without the Rust runtime and without a GPU.  It hashes with Strata's vendored xxhash (inlined: Strata links its own).
set(SEABRAID_XXHASH_DIR "${STRATA_SOURCE}/third_party/strata-llama/vendor/hash/xxhash")
if(NOT EXISTS "${SEABRAID_XXHASH_DIR}/xxhash.h")
  message(FATAL_ERROR "file moves need Strata's vendored xxhash.h (${SEABRAID_XXHASH_DIR}); STRATA_SOURCE is not a Strata checkout")
endif()
add_library(seabraid_strata_file_moves STATIC file_moves.cpp engine_flags.cpp)
set_target_properties(seabraid_strata_file_moves PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(seabraid_strata_file_moves PUBLIC cxx_std_17)
target_include_directories(seabraid_strata_file_moves PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${SEABRAID_STRATA_WIRE}")
target_include_directories(seabraid_strata_file_moves SYSTEM PUBLIC "${STRATA_SOURCE}/include")
target_include_directories(seabraid_strata_file_moves SYSTEM PRIVATE "${SEABRAID_XXHASH_DIR}")
target_link_libraries(seabraid_strata_file_moves PUBLIC seabraid_strata_diag)
seabraid_harden(seabraid_strata_file_moves LIB)

# The routing table depends on diag only, so it builds without Strata or the Rust runtime.
add_library(seabraid_strata_executor_view STATIC executor_view.cpp)
set_target_properties(seabraid_strata_executor_view PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(seabraid_strata_executor_view PUBLIC cxx_std_17)
target_include_directories(seabraid_strata_executor_view PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(seabraid_strata_executor_view PUBLIC seabraid_strata_diag Threads::Threads)
seabraid_harden(seabraid_strata_executor_view LIB)

add_library(seabraid_strata_static_qwen STATIC static_qwen.cpp static_qwen_tier.cpp private_qwen.cpp spawn_policy.cpp
                                          helper_process.cpp helper_process_stop.cpp helper_exit.cpp helper_kernel.cpp)
seabraid_harden(seabraid_strata_static_qwen LIB)
if(SEABRAID_HAVE_PIDFD_SPAWN)
  # glibc >= 2.39: spawn and obtain the pidfd atomically (helper_process.cpp).
  target_compile_definitions(seabraid_strata_static_qwen PRIVATE SEABRAID_HAVE_PIDFD_SPAWN=1)
endif()
# The embedding host may include the adapter in a shared library.
set_target_properties(seabraid_strata_static_qwen PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(seabraid_strata_static_qwen PUBLIC cxx_std_17)
target_include_directories(seabraid_strata_static_qwen PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${SEABRAID_STRATA_WIRE}" "../../include")
# Share the helper's build configuration with any consumer compiling the same headers.
if(STRATA_GEOMETRY_INCLUDE)
  target_include_directories(seabraid_strata_static_qwen SYSTEM PUBLIC "${STRATA_GEOMETRY_INCLUDE}")
endif()
target_include_directories(seabraid_strata_static_qwen SYSTEM PUBLIC "${STRATA_SOURCE}/include")
if(STRATA_WINDOW_MAX_T AND NOT STRATA_WINDOW_MAX_T STREQUAL "auto")
  target_compile_definitions(seabraid_strata_static_qwen PUBLIC "STRATA_VERIFY_MAX_T=${STRATA_WINDOW_MAX_T}")
endif()
# Named shm functions live in librt on older glibc releases.
find_package(Threads REQUIRED)
target_link_libraries(seabraid_strata_static_qwen PUBLIC seabraid_strata_diag rt Threads::Threads)
